cmake_minimum_required(VERSION 3.20)
project(ForestHomeDistribution LANGUAGES C CXX)
enable_testing()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(FOREST_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/Forest_Home_Full_Project.zip")
set(FOREST_EXTRACTED "${CMAKE_BINARY_DIR}/source/final_gra")
file(SHA256 "${FOREST_ARCHIVE}" FOREST_ARCHIVE_SHA)
set(FOREST_PREVIOUS_SHA "")
if(EXISTS "${CMAKE_BINARY_DIR}/source/archive.sha256")
  file(READ "${CMAKE_BINARY_DIR}/source/archive.sha256" FOREST_PREVIOUS_SHA)
endif()
if(NOT FOREST_ARCHIVE_SHA STREQUAL FOREST_PREVIOUS_SHA OR NOT EXISTS "${FOREST_EXTRACTED}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${FOREST_ARCHIVE}" DESTINATION "${CMAKE_BINARY_DIR}/source")
  file(WRITE "${CMAKE_BINARY_DIR}/source/archive.sha256" "${FOREST_ARCHIVE_SHA}")
endif()
add_subdirectory("${FOREST_EXTRACTED}" "${CMAKE_BINARY_DIR}/app")
